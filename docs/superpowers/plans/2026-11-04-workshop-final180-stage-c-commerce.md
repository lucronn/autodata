```python
import json
import unittest
from scripts.dev.workshop_final180_identity_adapter import AdapterError, SimulatedExit
from scripts.dev import workshop_final180_commerce_adapter as adapter

def clean_xml():
    commerce = ''.join('<testcase classname="tests.test_commerce_postgres" name="' + name + '"/>'
                       for name in COMMERCE_NAMES)
    support = ''.join('<testcase classname="synthetic.unobserved" name="' + name + '"/>'
                      for name in SUPPORT_NAMES)
    return ('<testsuite>' + commerce + support + '</testsuite>').encode('utf-8')

def clean_go():
    package = 'github.com/lucronn/autodata/apps/api-go'
    root = 'TestWorkshopCommercePythonIntegration'
    events = [dict(Time='synthetic', Action='start', Package=package),
              dict(Time='synthetic', Action='run', Package=package, Test=root),
              dict(Time='synthetic', Action='pass', Package=package, Test=root),
              dict(Time='synthetic', Action='pass', Package=package)]
    return b'\n'.join(json.dumps(e, separators=(',', ':')).encode() for e in events) + b'\n'

def context():
    return adapter.CommerceContext('4d82e413e04fb2401ac7bc28dc828375ac662180',
        'c4e9293875d22e04469c6a3fb4cdab27889328d6519243cee3862f9ecf490b39', '0' * 64)

def translate(pytest_receipt, go_raw, go_receipt):
    support = adapter.CommerceGoSupport(go_raw, b'', go_receipt)
    value = adapter.CommerceInput(clean_xml(), b'', b'', pytest_receipt, support)
    return adapter.translate_commerce(value, context())

class CommerceTranslationTests(unittest.TestCase):
    def test_clean_synthetic_rows(self):
        clean = SimulatedExit(0, False, False, False, True)
        result = translate(clean, clean_go(), clean)
        self.assertEqual(result.status, 'consistent')
        self.assertEqual(result.reasons, ())
        self.assertEqual(tuple(row.logical_id for row in result.observations), COMMERCE_NAMES)
        self.assertEqual(tuple(row.status for row in result.observations), ('passed',) * 14)
        self.assertIs(type(result.accepted_case_credit), int)
        self.assertEqual(result.accepted_case_credit, 0)
        self.assertTrue(all(value is False for _, value in result.authority))

    def test_mixed_failure_retains_both_stream_facts(self):
        clean = SimulatedExit(0, False, False, False, True)
        result = translate(SimulatedExit(1, False, False, False, True), b'{\n', clean)
        self.assertEqual(result.status, 'unavailable')
        self.assertEqual(result.reasons, ('go:json_invalid', 'pytest:simulated_nonzero'))
        self.assertEqual(tuple(row.status for row in result.observations),
                         ('unavailable',) + ('failed',) * 13)
```
